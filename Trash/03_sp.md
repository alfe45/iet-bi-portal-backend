
----------------------------------------------------------------------------
-- ASIGNATURAS
----------------------------------------------------------------------------
CREATE OR REPLACE PROCEDURE academico.registrar_asignatura(
	IN p_codigo VARCHAR(3),
	IN p_tipo_asignatura academico.tipo_asignatura,
	IN p_nombre VARCHAR(100),
	IN p_descripcion VARCHAR(255),
	OUT p_id_asignatura BIGINT
)
LANGUAGE plpgsql
AS $$
BEGIN
	-- ✅
	IF NULLIF(trim(p_codigo), '') IS NULL
		OR p_tipo_asignatura IS NULL
		OR NULLIF(trim(p_nombre), '') IS NULL THEN
		PERFORM academico.fn_lanzar_excepcion(
			'El código, el tipo y el nombre de la asignatura son obligatorios.',
			jsonb_build_object(
				'codigo', p_codigo,
				'tipo_asignatura', p_tipo_asignatura,
				'nombre', p_nombre
			),
			'Complete todos los datos requeridos de la asignatura.',
			'AS001'
		);
	END IF;

	-- ✅
	IF length(trim(p_codigo)) <> 3 THEN
		PERFORM academico.fn_lanzar_excepcion(
			'El código de la asignatura debe tener exactamente 3 caracteres.',
			jsonb_build_object('codigo', p_codigo),
			NULL,
			'AS002'
		);
	END IF;

	-- ✅
	IF length(trim(p_nombre)) < 2 THEN
		PERFORM academico.fn_lanzar_excepcion(
			'El nombre de la asignatura debe tener al menos 2 caracteres.',
			jsonb_build_object('nombre', p_nombre),
			NULL,
			'AS003'
		);
	END IF;

	-- ✅
	IF EXISTS (
		SELECT 1
		FROM academico.asignaturas
		WHERE codigo = trim(p_codigo)
	) THEN
		PERFORM academico.fn_lanzar_excepcion(
			'Ya existe una asignatura con ese código.',
			jsonb_build_object('codigo', p_codigo),
			NULL,
			'AS004'
		);
	END IF;

	-- ✅
	IF EXISTS (
		SELECT 1
		FROM academico.asignaturas
		WHERE lower(trim(nombre)) = lower(trim(p_nombre))
	) THEN
		PERFORM academico.fn_lanzar_excepcion(
			'Ya existe una asignatura con ese nombre.',
			jsonb_build_object('nombre', p_nombre),
			NULL,
			'AS005'
		);
	END IF;

	INSERT INTO academico.asignaturas (
		codigo,
		tipo_asignatura,
		nombre,
		descripcion
	)
	VALUES (
		NULLIF(trim(p_codigo), ''),
		p_tipo_asignatura,
		NULLIF(trim(p_nombre), ''),
		NULLIF(trim(p_descripcion), '')
	)
	RETURNING id_asignatura INTO p_id_asignatura;
END;
$$;
